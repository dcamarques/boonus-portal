// --- CONTROLE DE PERFIL, CONVITES E ONBOARDING NO DASHBOARD --- let convitesPendentesCache = []; async function configurarCabecalhoEOnboarding(session, perfil) { const nomeExibicao = perfil?.nome || session.user.user_metadata?.nome || session.user.email; const cargoExibicao = perfil?.cargo || (perfil?.nivel_acesso === 1 ? 'Sócio Diretor' : 'Membro'); const elOla = document.getElementById('user-ola'); const elCargo = document.getElementById('user-cargo'); if (elOla) elOla.textContent = `Olá, ${nomeExibicao}`; if (elCargo) elCargo.textContent = cargoExibicao; await carregarConvitesPendentes(session.user.email); if (!perfil?.escritorio_id) { if (convitesPendentesCache.length > 0) { await abrirModalConvites(); } else { const res = await Swal.fire({ title: 'Deseja cadastrar seu escritório agora?', text: 'Você pode cadastrar seu escritório agora ou pular e explorar o painel.', icon: 'info', showCancelButton: true, confirmButtonText: 'Sim, cadastrar escritório', cancelButtonText: 'Pular por enquanto', confirmButtonColor: '#00ff88', cancelButtonColor: '#333', background: '#111', color: '#fff' }); if (res.isConfirmed) { await abrirModalCadastrarEscritorio(session, perfil); } } } } async function carregarConvitesPendentes(emailUsuario) { if (!emailUsuario) return; const { data, error } = await supabaseClient .from('ES_convites') .select('*, ES_escritorios(nome)') .ilike('email_convidado', emailUsuario.trim()) .eq('status', 'pendente'); convitesPendentesCache = error ? [] : (data || []); const badge = document.getElementById('badge-convites'); if (badge) { badge.textContent = convitesPendentesCache.length > 0 ? ` (${convitesPendentesCache.length})` : ''; } } function exigirEscritorioOuConvite(acaoCallback) { if (sessaoAtual && sessaoAtual.escritorio_id) { if (typeof acaoCallback === 'function') acaoCallback(); return true; } Swal.fire({ title: 'Acesso pendente', text: 'Cadastre o seu escritório ou veja se você possui algum convite.', icon: 'warning', showCancelButton: true, showDenyButton: true, confirmButtonText: 'Cadastrar Escritório', denyButtonText: 'Ver Meus Convites', cancelButtonText: 'Fechar', confirmButtonColor: '#00ff88', denyButtonColor: '#2563eb', cancelButtonColor: '#333', background: '#111', color: '#fff' }).then(async (result) => { if (result.isConfirmed) { await abrirModalCadastrarEscritorio(sessaoAtual.session, sessaoAtual.perfil); } else if (result.isDenied) { await abrirModalConvites(); } }); return false; } async function abrirModalCadastrarEscritorio(session, perfil) { const { value: formValues } = await Swal.fire({ title: 'Cadastrar Meu Escritório', html: `
`, background: '#111', color: '#fff', confirmButtonColor: '#00ff88', preConfirm: () => { const nome = document.getElementById('swal-cad-esc-nome').value.trim(); if (!nome) { Swal.showValidationMessage('Informe o nome do escritório.'); return null; } return { nome, documento: document.getElementById('swal-cad-esc-doc').value.trim() || null }; } }); if (!formValues) return; const { data: escCriado, error: errEsc } = await supabaseClient .from('ES_escritorios') .insert([{ nome: formValues.nome, documento: formValues.documento, ativo: true }]) .select() .single(); if (errEsc) { Swal.fire('Erro', errEsc.message, 'error'); return; } await supabaseClient .from('ES_master_escritorios') .insert([{ master_auth_id: session.user.id, escritorio_id: escCriado.id }]); const { error: errUpd } = await supabaseClient .from('ES_usuarios_perfis') .update({ escritorio_id: escCriado.id, nivel_acesso: 1, cargo: perfil?.cargo || 'Sócio Diretor' }) .eq('auth_id', session.user.id); if (errUpd) { Swal.fire('Erro', errUpd.message, 'error'); return; } await Swal.fire({ icon: 'success', title: 'Escritório cadastrado!', text: `O escritório "${escCriado.nome}" já está ativo na sua conta Master.`, background: '#111', color: '#fff', confirmButtonColor: '#00ff88' }); window.location.reload(); } async function abrirModalConvites() { const emailAtual = sessaoAtual?.session?.user?.email; await carregarConvitesPendentes(emailAtual); if (convitesPendentesCache.length === 0) { Swal.fire({ icon: 'info', title: 'Sem convites pendentes', text: 'Não há convites pendentes para o seu e-mail no momento. Você pode cadastrar o seu próprio escritório.', background: '#111', color: '#fff', confirmButtonColor: '#00ff88' }); return; } const convite = convitesPendentesCache[0]; const nomeEsc = convite.ES_escritorios?.nome || 'Escritório Convidado'; const nivelTexto = convite.nivel_acesso === 2 ? 'Nível 2 (Responsável)' : 'Nível 3 (Assessor)'; const res = await Swal.fire({ title: 'Convite de Escritório', html: `

Escritório: ${nomeEsc}

Cargo: ${convite.cargo || 'Assessor'}

Nível: ${nivelTexto}

`, icon: 'question', showCancelButton: true, confirmButtonText: 'Aceitar Convite', cancelButtonText: 'Recusar', confirmButtonColor: '#00ff88', cancelButtonColor: '#ff4444', background: '#111', color: '#fff' }); if (res.isConfirmed) { await supabaseClient .from('ES_usuarios_perfis') .update({ escritorio_id: convite.escritorio_id, nivel_acesso: convite.nivel_acesso, cargo: convite.cargo || 'Assessor', ativo: true }) .eq('auth_id', sessaoAtual.session.user.id); await supabaseClient .from('ES_convites') .update({ status: 'aceito' }) .eq('id', convite.id); await Swal.fire({ icon: 'success', title: 'Convite aceito!', text: `Agora você faz parte do escritório ${nomeEsc}.`, background: '#111', color: '#fff', confirmButtonColor: '#00ff88' }); window.location.reload(); } else if (res.dismiss === Swal.DismissReason.cancel) { await supabaseClient .from('ES_convites') .update({ status: 'recusado' }) .eq('id', convite.id); await carregarConvitesPendentes(emailAtual); } }
